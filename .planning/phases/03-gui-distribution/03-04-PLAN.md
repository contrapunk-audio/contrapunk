3. Implement the show() method to draw the keyboard:
   ```rust
   impl PianoKeyboard {
       pub fn show(&self, ui: &mut Ui) {
           let white_key_width = 24.0;
           let white_key_height = 100.0;
           let black_key_width = 14.0;
           let black_key_height = 60.0;

           // Calculate total width (7 white keys per octave)
           let total_white_keys = self.octaves as usize * 7;
           let total_width = total_white_keys as f32 * white_key_width;

           let (response, painter) = ui.allocate_painter(
               Vec2::new(total_width, white_key_height),
               Sense::hover(),
           );
           let rect = response.rect;

           // Draw white keys first
           self.draw_white_keys(&painter, rect, white_key_width, white_key_height);

           // Draw black keys on top
           self.draw_black_keys(&painter, rect, white_key_width, black_key_width, black_key_height);
       }
   }
   ```

4. Implement helper methods for drawing keys:
   ```rust
   fn draw_white_keys(&self, painter: &egui::Painter, rect: Rect, key_width: f32, key_height: f32) {
       let white_key_indices = [0, 2, 4, 5, 7, 9, 11]; // C, D, E, F, G, A, B offsets

       for octave in 0..self.octaves as usize {
           for (i, &offset) in white_key_indices.iter().enumerate() {
               let midi = self.start_note + (octave as u8 * 12) + offset;
               let x = rect.left() + (octave * 7 + i) as f32 * key_width;

               let key_rect = Rect::from_min_size(
                   egui::pos2(x, rect.top()),
                   Vec2::new(key_width - 1.0, key_height),
               );

               let color = self.key_color(midi, false);
               painter.rect_filled(key_rect, 2.0, color);
               painter.rect_stroke(key_rect, 2.0, egui::Stroke::new(1.0, Color32::DARK_GRAY));
           }
       }
   }

   fn draw_black_keys(&self, painter: &egui::Painter, rect: Rect, white_width: f32, black_width: f32, black_height: f32) {
       // Black keys: C#, D#, F#, G#, A# (indices 1, 3, 6, 8, 10 in chromatic scale)
       // They sit between white keys at positions: after C, D, F, G, A
       let black_positions = [(0, 1), (1, 3), (3, 6), (4, 8), (5, 10)]; // (white_key_index, chromatic_offset)

       for octave in 0..self.octaves as usize {
           for &(white_idx, offset) in &black_positions {
               let midi = self.start_note + (octave as u8 * 12) + offset;
               let x = rect.left() + (octave * 7 + white_idx) as f32 * white_width + white_width - black_width / 2.0;

               let key_rect = Rect::from_min_size(
                   egui::pos2(x, rect.top()),
                   Vec2::new(black_width, black_height),
               );

               let color = self.key_color(midi, true);
               painter.rect_filled(key_rect, 2.0, color);
           }
       }
   }

   fn key_color(&self, midi: u8, is_black: bool) -> Color32 {
       let is_input = self.input_notes.contains(&midi);
       let is_harmony = self.harmony_notes.contains(&midi);

       match (is_input, is_harmony) {
           (true, true) => Color32::from_rgb(0, 200, 200),   // Cyan - both
           (true, false) => Color32::from_rgb(100, 149, 237), // Blue - input only
           (false, true) => Color32::from_rgb(50, 205, 50),   // Green - harmony only
           (false, false) => {
               if is_black {
                   Color32::from_rgb(30, 30, 30)  // Dark gray for black keys
               } else {
                   Color32::from_rgb(250, 250, 250)  // White for white keys
               }
           }
       }
   }
   ```
