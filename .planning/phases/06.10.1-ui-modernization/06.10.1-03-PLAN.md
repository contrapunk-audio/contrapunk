```typescript
export interface MidiDevice {
  index: number;
  name: string;
}

export interface EngineState {
  key: string;
  mode: string;
  scaleMode: string;
  octaveMode: string;
  voiceLeadingEnabled: boolean;
  voiceLeadingStyle: string;
  interchangeEnabled: boolean;
  interchangeRange: number;
  voicePosition: number;
  isRunning: boolean;
}

export interface NoteState {
  inputNotes: number[];
  harmonyNotes: number[];
  borrowedNotes: number[];
  chordName: string;
  lastBorrowedFrom: string;
}

export interface HumanizeState {
  enabled: boolean;
  timingJitter: number;
  velocityVariation: number;
  durationVariation: number;
  swingAmount: number;
  bpm: number;
  metronomeEnabled: boolean;
}

export interface Preset {
  name: string;
  isBuiltin: boolean;
}

export interface ContrapunkAdapter {
  // Initialization
  init(): Promise<void>;

  // Engine state
  getEngineState(): Promise<EngineState>;
  setKey(key: string): Promise<void>;
  setMode(mode: string): Promise<void>;
  setScaleMode(mode: string): Promise<void>;
  setOctaveMode(mode: string): Promise<void>;
  setVoiceLeading(enabled: boolean, style: string): Promise<void>;
  setInterchange(enabled: boolean, range: number): Promise<void>;
  setVoicePosition(position: number): Promise<void>;

  // Humanization
  getHumanizeState(): Promise<HumanizeState>;
  setHumanizeConfig(config: Partial<HumanizeState>): Promise<void>;

  // MIDI devices
  listMidiInputs(): Promise<MidiDevice[]>;
  listMidiOutputs(): Promise<MidiDevice[]>;
  refreshMidiDevices(): Promise<void>;

  // Routing
  startRouting(inputIdx: number, outputIndices: number[]): Promise<void>;
  stopRouting(): Promise<void>;

  // Real-time state
  getNoteState(): Promise<NoteState>;
  onNoteUpdate(callback: (state: NoteState) => void): () => void;

  // Presets
  listPresets(): Promise<Preset[]>;
  loadPreset(name: string): Promise<void>;
  savePreset(name: string): Promise<void>;
  deletePreset(name: string): Promise<void>;
}
```

**ui/src/lib/adapter/tauri.ts** — Tauri IPC implementation:
- Import `invoke` from `@tauri-apps/api/core` and `listen` from `@tauri-apps/api/event`
- Implement all ContrapunkAdapter methods using `invoke('command_name', { args })`
- For `onNoteUpdate`: use `listen('note-update', callback)` which returns an unlisten function
- Handle errors by catching invoke failures and rethrowing with user-friendly messages

**ui/src/lib/adapter/wasm.ts** — WASM implementation:
- Import from the wasm-pkg output (path TBD based on wasm-pack output location)
- Implement all methods by calling wasm-bindgen exported functions directly
- For `onNoteUpdate`: use a polling approach (requestAnimationFrame) since WASM doesn't push events
- `init()` must call the WASM init function and await it before any other calls

**ui/src/lib/adapter/index.ts** — Platform detection and export:
- `isTauri()`: Check `typeof window !== 'undefined' && '__TAURI__' in window`
- Export a singleton adapter: `export const adapter: ContrapunkAdapter = isTauri() ? new TauriAdapter() : new WasmAdapter();`
- Also export a `platformName` const: 'tauri' or 'browser' for display purposes

Note: Per CONTEXT.md Claude's discretion, we use the adapter pattern (not platform-specific code in components). This was the research recommendation.
