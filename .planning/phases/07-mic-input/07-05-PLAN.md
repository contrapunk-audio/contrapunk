```rust
#[cfg(not(target_arch = "wasm32"))]
fn start_mic_capture(&mut self) -> Result<(), String> {
    use crate::audio::{get_input_device, default_input_device, AudioCapture};

    // Get the audio device
    let device = if let Some(idx) = self.mic_state.selected_device {
        get_input_device(idx).map_err(|e| e.to_string())?
    } else {
        default_input_device().map_err(|e| e.to_string())?
    };

    // Create audio capture with 8192 sample ring buffer (~185ms at 44.1kHz)
    let capture = AudioCapture::new(&device, 8192)
        .map_err(|e| e.to_string())?;

    let sample_rate = capture.sample_rate() as usize;

    // Create pitch detector
    let pitch_detector = PitchDetector::new(sample_rate, &self.mic_state.config);

    // Reset note tracker
    self.mic_state.note_tracker = NoteTracker::new(
        self.mic_state.config.hysteresis_cents,
        self.mic_state.config.min_note_duration_ms,
    );

    self.mic_state.capture = Some(capture);
    self.mic_state.pitch_detector = Some(pitch_detector);
    self.mic_state.is_active = true;

    Ok(())
}
```

3. Add process_mic_frame() for native:
```rust
#[cfg(not(target_arch = "wasm32"))]
fn process_mic_frame(&mut self) {
    if !self.mic_state.is_active {
        return;
    }

    let Some(ref mut capture) = self.mic_state.capture else { return };
    let Some(ref mut detector) = self.mic_state.pitch_detector else { return };

    // Read available samples
    let mut buffer = vec![0.0f32; 4096];
    let samples_read = capture.read_samples(&mut buffer);

    if samples_read == 0 {
        return;
    }

    // Add samples to detector
    detector.add_samples(&buffer[..samples_read]);

    // Update RMS for level meter
    self.mic_state.current_rms = detector.last_rms();

    // Try to detect pitch
    let pitch = detector.detect();
    self.mic_state.last_pitch = pitch.clone();

    // Get current time
    let now_ms = std::time::Instant::now().elapsed().as_millis() as f64;
    // Note: Need a start time reference. For simplicity, use a simple counter.
    // In practice, track start time when capture begins.

    // Update note tracker
    if let Some(event) = self.mic_state.note_tracker.update(pitch.as_ref(), now_ms) {
        self.handle_mic_note_event(event);
    }
}

fn handle_mic_note_event(&mut self, event: NoteEvent) {
    use wmidi::{Note, Channel, Velocity};

    match event {
        NoteEvent::NoteOn(midi) => {
            let note = Note::from_u8_lossy(midi);
            let velocity = Velocity::try_from(100u8).unwrap();
            // Process through harmony engine (similar to MIDI input)
            self.process_mic_note_on(note, velocity);
        }
        NoteEvent::NoteOff(midi) => {
            let note = Note::from_u8_lossy(midi);
            self.process_mic_note_off(note);
        }
        NoteEvent::NoteChange { old, new } => {
            // Note off for old, note on for new
            let old_note = Note::from_u8_lossy(old);
            self.process_mic_note_off(old_note);
            let new_note = Note::from_u8_lossy(new);
            let velocity = Velocity::try_from(100u8).unwrap();
            self.process_mic_note_on(new_note, velocity);
        }
    }
}
```
