2. Create src/audio/pitch.rs with DetectedPitch and freq_to_midi:
   ```rust
   /// Result of pitch detection.
   #[derive(Clone, Debug)]
   pub struct DetectedPitch {
       /// Detected frequency in Hz
       pub frequency: f32,
       /// Nearest MIDI note number (0-127)
       pub midi_note: u8,
       /// Deviation from nearest note in cents (-50 to +50)
       pub cents_deviation: i8,
       /// Confidence/clarity score (0.0-1.0)
       pub clarity: f32,
   }

   /// Converts frequency (Hz) to MIDI note with cents deviation.
   /// Formula: midi = 69 + 12 * log2(freq / 440)
   pub fn freq_to_midi(freq: f32) -> (u8, i8) {
       if freq <= 0.0 {
           return (0, 0);
       }
       let midi_float = 69.0 + 12.0 * (freq / 440.0).log2();
       let midi_note = midi_float.round() as i32;
       let cents = ((midi_float - midi_note as f32) * 100.0) as i8;
       (midi_note.clamp(0, 127) as u8, cents.clamp(-50, 50))
   }

   /// Checks if a MIDI note is within the voice range (C2-C6).
   pub fn is_in_voice_range(midi_note: u8, min: u8, max: u8) -> bool {
       midi_note >= min && midi_note <= max
   }
   ```

3. Add NoteEvent enum and NoteTracker struct to pitch.rs:
   ```rust
   /// Events emitted by the note tracker.
   #[derive(Clone, Debug)]
   pub enum NoteEvent {
       /// New note started
       NoteOn(u8),
       /// Note ended (silence or pitch change)
       NoteOff(u8),
       /// Pitch changed from old to new note
       NoteChange { old: u8, new: u8 },
   }

   /// Tracks pitch over time with hysteresis to prevent vibrato triggering note changes.
   pub struct NoteTracker {
       current_note: Option<u8>,
       note_start_ms: f64,
       hysteresis_cents: i8,
       min_duration_ms: u32,
   }

   impl NoteTracker {
       pub fn new(hysteresis_cents: i8, min_duration_ms: u32) -> Self {
           Self {
               current_note: None,
               note_start_ms: 0.0,
               hysteresis_cents,
               min_duration_ms,
           }
       }

       /// Update with a new pitch detection. Returns any note event.
       pub fn update(&mut self, pitch: Option<&DetectedPitch>, now_ms: f64) -> Option<NoteEvent> {
           match (self.current_note, pitch) {
               // No current note, no detection -> nothing
               (None, None) => None,

               // No current note, new detection -> Note On
               (None, Some(p)) => {
                   self.current_note = Some(p.midi_note);
                   self.note_start_ms = now_ms;
                   Some(NoteEvent::NoteOn(p.midi_note))
               }

               // Current note, silence -> Note Off
               (Some(current), None) => {
                   self.current_note = None;
                   Some(NoteEvent::NoteOff(current))
               }

               // Current note, new detection -> check for change
               (Some(current), Some(p)) => {
                   if current == p.midi_note {
                       // Same note, no event
                       None
                   } else {
                       // Different note - check hysteresis and duration
                       let cents_diff = ((p.midi_note as i16 - current as i16) * 100
                           + p.cents_deviation as i16).abs();
                       let duration = now_ms - self.note_start_ms;

                       // Only change if beyond hysteresis AND minimum duration passed
                       if cents_diff > (self.hysteresis_cents as i16 * 2)
                           && duration >= self.min_duration_ms as f64
                       {
                           let old = current;
                           self.current_note = Some(p.midi_note);
                           self.note_start_ms = now_ms;
                           Some(NoteEvent::NoteChange { old, new: p.midi_note })
                       } else {
                           None
                       }
                   }
               }
           }
       }

       /// Force release current note (e.g., when stopping mic input).
       pub fn release(&mut self) -> Option<NoteEvent> {
           self.current_note.take().map(NoteEvent::NoteOff)
       }

       /// Get the currently active note, if any.
       pub fn current_note(&self) -> Option<u8> {
           self.current_note
       }
   }
   ```

   use super::config::MicConfig;

   /// A saved microphone configuration profile.
   #[derive(Clone, Debug, Serialize, Deserialize)]
   pub struct MicProfile {
       /// Profile name (e.g., "Quiet Studio", "Noisy Room")
       pub name: String,
       /// Microphone configuration
       pub config: MicConfig,
       /// Whether this is a built-in profile (cannot be deleted)
       pub is_builtin: bool,
   }

   impl MicProfile {
       /// Create a new custom profile.
       pub fn new(name: impl Into<String>, config: MicConfig) -> Self {
           Self {
               name: name.into(),
               config,
               is_builtin: false,
           }
       }

       /// Create a built-in profile.
       pub fn builtin(name: impl Into<String>, config: MicConfig) -> Self {
           Self {
               name: name.into(),
               config,
               is_builtin: true,
           }
       }
   }

   /// Returns the default built-in mic profiles.
   pub fn builtin_profiles() -> Vec<MicProfile> {
