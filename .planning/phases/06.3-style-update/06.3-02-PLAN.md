2. Create `src/preset/mod.rs`:
   ```rust
   pub mod builtins;
   pub mod storage;

   use serde::{Serialize, Deserialize};
   use crate::harmony::{Key, HarmonyMode, OctaveMode, VoiceLeadingStyle};
   use crate::humanize::HumanizeConfig;

   #[derive(Clone, Debug, Serialize, Deserialize)]
   pub struct StylePreset {
       pub name: String,
       pub persona: String,          // Character persona name
       pub genre: String,            // Genre subtitle
       pub harmony_mode: HarmonyMode,
       pub key: Key,
       pub voice_leading_enabled: bool,
       pub voice_leading_style: VoiceLeadingStyle,
       pub octave_mode: OctaveMode,
       pub humanize_config: HumanizeConfig,
       #[serde(skip)]
       pub is_builtin: bool,
   }

   pub struct PresetManager {
       builtins: Vec<StylePreset>,
       custom: Vec<StylePreset>,
       active_index: Option<usize>,  // index into combined list (builtins first, then custom)
   }
   ```
   PresetManager methods:
   - `new() -> Self` — loads builtins from builtins::all()
   - `all_presets(&self) -> Vec<&StylePreset>` — builtins + custom
   - `active(&self) -> Option<&StylePreset>`
   - `set_active(&mut self, index: usize)`
   - `add_custom(&mut self, preset: StylePreset)`
   - `remove_custom(&mut self, index: usize)` — only custom presets
   - `custom_presets(&self) -> &[StylePreset]`
   - `set_custom_presets(&mut self, presets: Vec<StylePreset>)`

3. Add `pub mod preset;` to src/lib.rs (feature-gated with `#[cfg(feature = "gui")]`).
