//! WASM audio capture using Web Audio API.
//!
//! Provides microphone input in the browser via getUserMedia and AnalyserNode.

use wasm_bindgen::prelude::*;
use wasm_bindgen::JsCast;
use web_sys::{
    AudioContext, AudioContextOptions, AnalyserNode, MediaStream,
    MediaStreamAudioSourceNode, MediaDevices, MediaStreamConstraints,
};

/// Result type for web audio operations.
pub type WebAudioResult<T> = Result<T, JsValue>;

/// Request microphone access from the browser.
///
/// IMPORTANT: This MUST be called from a user gesture (button click, etc.)
/// or the browser will reject the permission request.
///
/// Returns the MediaStream on success.
pub async fn request_mic_access() -> WebAudioResult<MediaStream> {
    let window = web_sys::window().ok_or_else(|| JsValue::from_str("No window"))?;
    let navigator = window.navigator();
    let media_devices: MediaDevices = navigator.media_devices()
        .map_err(|_| JsValue::from_str("No media devices"))?;

    // Create constraints for audio only
    let constraints = MediaStreamConstraints::new();
    constraints.set_audio(&JsValue::TRUE);
    constraints.set_video(&JsValue::FALSE);

    let promise = media_devices.get_user_media_with_constraints(&constraints)
        .map_err(|_| JsValue::from_str("getUserMedia not available"))?;

    let stream = wasm_bindgen_futures::JsFuture::from(promise).await?;
    stream.dyn_into::<MediaStream>()
}

/// List available audio input devices.
///
/// Returns a list of (device_id, label) tuples.
/// Note: Labels may be empty strings if permission hasn't been granted yet.
pub async fn list_audio_inputs() -> WebAudioResult<Vec<(String, String)>> {
    let window = web_sys::window().ok_or_else(|| JsValue::from_str("No window"))?;
    let navigator = window.navigator();
    let media_devices: MediaDevices = navigator.media_devices()
        .map_err(|_| JsValue::from_str("No media devices"))?;

    let promise = media_devices.enumerate_devices()
        .map_err(|_| JsValue::from_str("enumerate_devices not available"))?;

    let devices = wasm_bindgen_futures::JsFuture::from(promise).await?;
    let devices = js_sys::Array::from(&devices);

    let mut result = Vec::new();
    for device in devices.iter() {
        if let Ok(info) = device.dyn_into::<web_sys::MediaDeviceInfo>() {
            if info.kind() == web_sys::MediaDeviceKind::Audioinput {
                let id = info.device_id();
                let label = info.label();
                result.push((id, if label.is_empty() { "Microphone".to_string() } else { label }));
            }
        }
    }

    Ok(result)
}

/// Web Audio capture state for WASM builds.
pub struct WebAudioCapture {
    /// Audio context for Web Audio API
    context: AudioContext,
    /// Media stream source node
    _source: MediaStreamAudioSourceNode,
    /// Analyser node for getting audio data
    analyser: AnalyserNode,
    /// FFT size (determines buffer size)
    fft_size: u32,
    /// Temporary buffer for reading time domain data
    time_data: Vec<f32>,
}

impl WebAudioCapture {
    /// Create a new WebAudioCapture from a MediaStream.
    ///
    /// # Arguments
    /// * `stream` - MediaStream from getUserMedia
    /// * `buffer_size` - Desired buffer size (will be rounded up to power of 2)
    pub fn new(stream: &MediaStream, buffer_size: usize) -> WebAudioResult<Self> {
        // Create audio context
        let mut options = AudioContextOptions::new();
        options.sample_rate(44100.0); // Request 44.1kHz

        let context = AudioContext::new_with_context_options(&options)
            .map_err(|_| JsValue::from_str("Failed to create AudioContext"))?;

        // Create source from media stream
        let source = context.create_media_stream_source(stream)
            .map_err(|_| JsValue::from_str("Failed to create media stream source"))?;

        // Create analyser node
        let analyser = context.create_analyser()
            .map_err(|_| JsValue::from_str("Failed to create analyser"))?;

        // Round buffer_size up to power of 2 (AnalyserNode requires power of 2)
        let fft_size = (buffer_size as u32).next_power_of_two().max(256).min(32768);
        analyser.set_fft_size(fft_size);

        // Connect source to analyser (no need to connect to destination for input)
        source.connect_with_audio_node(&analyser)
            .map_err(|_| JsValue::from_str("Failed to connect source to analyser"))?;

        let time_data = vec![0.0f32; fft_size as usize];

        Ok(Self {
            context,
            _source: source,
            analyser,
            fft_size,
            time_data,
        })
    }

    /// Read current audio samples into the provided buffer.
    ///
    /// Returns the number of samples read.
    pub fn read_samples(&mut self, buffer: &mut [f32]) -> usize {
        // Get time domain data as bytes (getFloatTimeDomainData requires Float32Array)
        // Using getByteTimeDomainData for simplicity, then convert
        let mut byte_data = vec![0u8; self.fft_size as usize];
        self.analyser.get_byte_time_domain_data(&mut byte_data);

        // Convert from unsigned byte (0-255, center at 128) to float (-1.0 to 1.0)
        let count = buffer.len().min(byte_data.len());
        for (i, &byte) in byte_data.iter().take(count).enumerate() {
            buffer[i] = (byte as f32 - 128.0) / 128.0;
        }

        count
    }

    /// Read current audio samples as f32 directly.
    ///
    /// More accurate than read_samples but requires js_sys::Float32Array.
    pub fn read_samples_f32(&mut self) -> &[f32] {
        // Create a Float32Array view into our buffer
        let array = js_sys::Float32Array::new_with_length(self.fft_size);
        self.analyser.get_float_time_domain_data(&array);

        // Copy to our buffer
        array.copy_to(&mut self.time_data);
        &self.time_data
    }

    /// Returns the sample rate of the audio context.
    pub fn sample_rate(&self) -> f32 {
        self.context.sample_rate()
    }

    /// Returns the FFT/buffer size.
    pub fn buffer_size(&self) -> usize {
        self.fft_size as usize
    }

    /// Resume audio context (required after user gesture).
    pub async fn resume(&self) -> WebAudioResult<()> {
        let promise = self.context.resume()
            .map_err(|_| JsValue::from_str("Failed to resume audio context"))?;
        wasm_bindgen_futures::JsFuture::from(promise).await?;
        Ok(())
    }

    /// Returns whether the audio context is running.
    pub fn is_running(&self) -> bool {
        self.context.state() == web_sys::AudioContextState::Running
    }
}

/// Calculate RMS level from samples (for volume display).
pub fn calculate_rms(samples: &[f32]) -> f32 {
    if samples.is_empty() {
        return 0.0;
    }
    let sum_squares: f32 = samples.iter().map(|&s| s * s).sum();
    (sum_squares / samples.len() as f32).sqrt()
}

//! Audio capture and pitch detection module.
//!
//! Provides microphone input with real-time pitch detection for
//! audio-to-MIDI conversion.
//!
//! # Architecture
//!
//! ## Native (cpal)
//! ```text
//! [Microphone] -> [AudioCapture] -> [Ring Buffer] -> [PitchDetector] -> [NoteTracker] -> [MIDI Events]
//!                    (audio thread)                     (main thread)
//! ```
//!
//! ## WASM (Web Audio API)
//! ```text
//! [Microphone] -> [getUserMedia] -> [AnalyserNode] -> [PitchDetector] -> [NoteTracker] -> [MIDI Events]
//!                                   (in update() loop)
//! ```
//!
//! The key difference is that WASM uses AnalyserNode polling in the frame loop
//! rather than a separate audio thread with ring buffer.

pub mod config;
pub mod profiles;
pub mod pitch;
pub mod detector;

#[cfg(not(target_arch = "wasm32"))]
pub mod capture;

#[cfg(target_arch = "wasm32")]
pub mod web;

pub use config::MicConfig;
pub use profiles::{MicProfile, builtin_profiles};
pub use pitch::{DetectedPitch, NoteEvent, NoteTracker, freq_to_midi, is_in_voice_range};
pub use detector::PitchDetector;

#[cfg(not(target_arch = "wasm32"))]
pub use capture::{
    AudioCapture, list_audio_inputs, get_input_device, default_input_device,
    calculate_rms, rms_to_db, level_for_display,
};

#[cfg(target_arch = "wasm32")]
pub use web::{
    WebAudioCapture, request_mic_access, list_audio_inputs as list_audio_inputs_web,
    calculate_rms as calculate_rms_web,
};
