//! Native audio capture using cpal.
//!
//! Provides low-latency audio input via platform audio APIs (CoreAudio, WASAPI, ALSA).
//! Audio samples are transferred from the audio callback to the main thread using
//! a lock-free ring buffer to prevent audio glitches.

use anyhow::{anyhow, Result};
use cpal::traits::{DeviceTrait, HostTrait, StreamTrait};
use cpal::{Device, Host, SampleFormat, Stream, SupportedStreamConfig};
use rtrb::{Consumer, Producer, RingBuffer};

/// Audio capture state including stream and sample consumer.
pub struct AudioCapture {
    /// The cpal audio stream (must be kept alive)
    _stream: Stream,
    /// Consumer end of the ring buffer for reading samples
    consumer: Consumer<f32>,
    /// Sample rate of the audio stream
    sample_rate: u32,
}

impl AudioCapture {
    /// Create a new audio capture from the specified device.
    ///
    /// # Arguments
    /// * `device` - The cpal input device to capture from
    /// * `buffer_capacity` - Ring buffer capacity in samples (e.g., 8192 for ~185ms at 44.1kHz)
    ///
    /// # Returns
    /// AudioCapture instance ready to provide samples via `read_samples()`.
    pub fn new(device: &Device, buffer_capacity: usize) -> Result<Self> {
        let config = device.default_input_config()
            .map_err(|e| anyhow!("Failed to get default input config: {}", e))?;

        let sample_rate = config.sample_rate().0;

        // Create ring buffer for lock-free transfer
        let (producer, consumer) = RingBuffer::<f32>::new(buffer_capacity);

        let stream = build_input_stream(device, &config, producer)?;
        stream.play().map_err(|e| anyhow!("Failed to start audio stream: {}", e))?;

        Ok(Self {
            _stream: stream,
            consumer,
            sample_rate,
        })
    }

    /// Read available samples into the provided buffer.
    ///
    /// Returns the number of samples read. Non-blocking.
    pub fn read_samples(&mut self, buffer: &mut [f32]) -> usize {
        let available = self.consumer.slots();
        let to_read = available.min(buffer.len());

        if to_read == 0 {
            return 0;
        }

        // Read samples from ring buffer
        match self.consumer.read_chunk(to_read) {
            Ok(chunk) => {
                let mut idx = 0;
                for sample in chunk.into_iter() {
                    if idx < buffer.len() {
                        buffer[idx] = sample;
                        idx += 1;
                    }
                }
                idx
            }
            Err(_) => 0,
        }
    }

    /// Returns the sample rate of the audio stream.
    pub fn sample_rate(&self) -> u32 {
        self.sample_rate
    }

    /// Returns the number of samples currently available to read.
    pub fn available_samples(&self) -> usize {
        self.consumer.slots()
    }
}

/// Build the cpal input stream with appropriate sample format handling.
fn build_input_stream(
    device: &Device,
    config: &SupportedStreamConfig,
    mut producer: Producer<f32>,
) -> Result<Stream> {
    let err_fn = |err| eprintln!("Audio stream error: {}", err);

    let stream = match config.sample_format() {
        SampleFormat::F32 => device.build_input_stream(
            &config.clone().into(),
            move |data: &[f32], _: &cpal::InputCallbackInfo| {
                // CRITICAL: This callback runs on audio thread.
                // NEVER block (no Mutex, no allocations, no channel send).
                // Only write to lock-free ring buffer.
                push_samples_to_ring_buffer(&mut producer, data);
            },
            err_fn,
            None,
        ),
        SampleFormat::I16 => device.build_input_stream(
            &config.clone().into(),
            move |data: &[i16], _: &cpal::InputCallbackInfo| {
                // Convert i16 to f32 and push
                // Note: This creates a temporary Vec which is technically an allocation,
                // but for typical buffer sizes (256-1024 samples) this is fast enough.
                // For production, consider a pre-allocated conversion buffer.
                let floats: Vec<f32> = data.iter()
                    .map(|&s| s as f32 / i16::MAX as f32)
                    .collect();
                push_samples_to_ring_buffer(&mut producer, &floats);
            },
            err_fn,
            None,
        ),
        SampleFormat::U16 => device.build_input_stream(
            &config.clone().into(),
            move |data: &[u16], _: &cpal::InputCallbackInfo| {
                let floats: Vec<f32> = data.iter()
                    .map(|&s| (s as f32 / u16::MAX as f32) * 2.0 - 1.0)
                    .collect();
                push_samples_to_ring_buffer(&mut producer, &floats);
            },
            err_fn,
            None,
        ),
        format => return Err(anyhow!("Unsupported sample format: {:?}", format)),
    }.map_err(|e| anyhow!("Failed to build input stream: {}", e))?;

    Ok(stream)
}

/// Push samples to ring buffer. Non-blocking - drops samples if buffer is full.
#[inline]
fn push_samples_to_ring_buffer(producer: &mut Producer<f32>, samples: &[f32]) {
    // Try to write all samples; if buffer is full, drop oldest by not writing
    // This is acceptable for real-time audio - we prefer dropping samples over blocking
    if let Ok(mut chunk) = producer.write_chunk(samples.len()) {
        for (slot, &sample) in chunk.as_mut_slices().0.iter_mut().zip(samples.iter()) {
            *slot = sample;
        }
        // write_chunk_uninit would be more efficient but write_chunk is safer
    }
    // If write_chunk fails (buffer full), samples are dropped - this is intentional
}

/// List available audio input devices.
///
/// Returns a list of (index, name) tuples for display in UI.
pub fn list_audio_inputs() -> Result<Vec<(usize, String)>> {
    let host = cpal::default_host();
    let devices: Vec<_> = host.input_devices()
        .map_err(|e| anyhow!("Failed to enumerate input devices: {}", e))?
        .collect();

    let mut result = Vec::new();
    for (idx, device) in devices.iter().enumerate() {
        let name = device.name().unwrap_or_else(|_| format!("Device {}", idx));
        result.push((idx, name));
    }

    Ok(result)
}

/// Get an audio input device by index.
pub fn get_input_device(index: usize) -> Result<Device> {
    let host = cpal::default_host();
    let devices: Vec<_> = host.input_devices()
        .map_err(|e| anyhow!("Failed to enumerate input devices: {}", e))?
        .collect();

    devices.into_iter()
        .nth(index)
        .ok_or_else(|| anyhow!("Audio device index {} out of range", index))
}

/// Get the default audio input device.
pub fn default_input_device() -> Result<Device> {
    cpal::default_host()
        .default_input_device()
        .ok_or_else(|| anyhow!("No default audio input device available"))
}

/// Calculate RMS (Root Mean Square) level of audio samples.
///
/// Returns a value between 0.0 (silence) and 1.0 (full scale).
pub fn calculate_rms(samples: &[f32]) -> f32 {
    if samples.is_empty() {
        return 0.0;
    }
    let sum_squares: f32 = samples.iter().map(|&s| s * s).sum();
    (sum_squares / samples.len() as f32).sqrt()
}

/// Convert RMS level to decibels.
///
/// Returns a value typically between -60 (silence) and 0 (full scale).
pub fn rms_to_db(rms: f32) -> f32 {
    if rms > 0.0 {
        20.0 * rms.log10()
    } else {
        -60.0 // Floor value for silence
    }
}

/// Convert RMS level to a 0.0-1.0 range for UI display.
///
/// Maps the range [floor_db, ceiling_db] to [0.0, 1.0].
pub fn level_for_display(rms: f32, floor_db: f32, ceiling_db: f32) -> f32 {
    let db = rms_to_db(rms);
    ((db - floor_db) / (ceiling_db - floor_db)).clamp(0.0, 1.0)
}
