//! Pitch detection using YIN algorithm.
//!
//! Wraps the pitch-detection crate's YIN detector with sample accumulation,
//! threshold filtering, and overlap processing for smooth pitch tracking.

use pitch_detection::detector::yin::YINDetector;
use pitch_detection::detector::PitchDetector as PitchDetectorTrait;

use super::config::MicConfig;
use super::pitch::{DetectedPitch, freq_to_midi, is_in_voice_range};

/// Pitch detector with sample accumulation and threshold filtering.
pub struct PitchDetector {
    /// YIN detector instance
    detector: YINDetector<f32>,
    /// Sample rate in Hz
    sample_rate: usize,
    /// Accumulation buffer for samples
    buffer: Vec<f32>,
    /// Buffer size for analysis (e.g., 2048)
    analysis_size: usize,
    /// Hop size for overlap (typically analysis_size / 2)
    hop_size: usize,
    /// Volume threshold (0.0-1.0)
    volume_threshold: f32,
    /// Confidence/clarity threshold (0.0-1.0)
    clarity_threshold: f32,
    /// Minimum MIDI note to detect
    min_note: u8,
    /// Maximum MIDI note to detect
    max_note: u8,
    /// Last RMS level for volume display
    last_rms: f32,
}

impl PitchDetector {
    /// Create a new pitch detector with the given configuration.
    ///
    /// # Arguments
    /// * `sample_rate` - Audio sample rate in Hz (e.g., 44100)
    /// * `config` - Microphone configuration with thresholds
    pub fn new(sample_rate: usize, config: &MicConfig) -> Self {
        let analysis_size = config.buffer_size;
        let hop_size = analysis_size / 2; // 50% overlap for smoother tracking
        let padding = analysis_size / 2;

        Self {
            detector: YINDetector::new(analysis_size, padding),
            sample_rate,
            buffer: Vec::with_capacity(analysis_size * 2),
            analysis_size,
            hop_size,
            volume_threshold: config.volume_threshold,
            clarity_threshold: config.confidence_threshold,
            min_note: config.min_note,
            max_note: config.max_note,
            last_rms: 0.0,
        }
    }

    /// Add samples to the internal buffer.
    pub fn add_samples(&mut self, samples: &[f32]) {
        self.buffer.extend_from_slice(samples);
    }

    /// Try to detect pitch from accumulated samples.
    ///
    /// Returns `Some(DetectedPitch)` if pitch detected with sufficient
    /// confidence, `None` otherwise. Automatically manages buffer.
    pub fn detect(&mut self) -> Option<DetectedPitch> {
        if self.buffer.len() < self.analysis_size {
            return None;
        }

        // Calculate RMS for volume gating and display
        let analysis_window = &self.buffer[..self.analysis_size];
        self.last_rms = calculate_rms(analysis_window);

        // Volume gate - reject if too quiet
        if self.last_rms < self.volume_threshold {
            // Remove processed samples (hop size for overlap)
            if self.buffer.len() >= self.hop_size {
                self.buffer.drain(..self.hop_size);
            }
            return None;
        }

        // Convert to f64 for pitch detection (library requires f64)
        let analysis_f64: Vec<f64> = analysis_window.iter()
            .map(|&s| s as f64)
            .collect();

        // Run YIN pitch detection
        let result = self.detector.get_pitch(
            &analysis_f64,
            self.sample_rate,
            self.volume_threshold as f64 * 10.0, // Power threshold (different scale)
            self.clarity_threshold as f64,
        );

        // Remove processed samples (hop size for overlap)
        if self.buffer.len() >= self.hop_size {
            self.buffer.drain(..self.hop_size);
        }

        // Process result
        let pitch = result?;

        // Convert frequency to MIDI
        let (midi_note, cents) = freq_to_midi(pitch.frequency as f32);

        // Filter by voice range
        if !is_in_voice_range(midi_note, self.min_note, self.max_note) {
            return None;
        }

        Some(DetectedPitch {
            frequency: pitch.frequency as f32,
            midi_note,
            cents_deviation: cents,
            clarity: pitch.clarity as f32,
        })
    }

    /// Get the last calculated RMS level (0.0-1.0).
    pub fn last_rms(&self) -> f32 {
        self.last_rms
    }

    /// Update configuration thresholds (can be called while running).
    pub fn update_config(&mut self, config: &MicConfig) {
        self.volume_threshold = config.volume_threshold;
        self.clarity_threshold = config.confidence_threshold;
        self.min_note = config.min_note;
        self.max_note = config.max_note;

        // Note: Changing buffer_size would require recreating the detector
        // and is not supported during operation.
    }

    /// Clear the internal buffer (e.g., when stopping).
    pub fn clear(&mut self) {
        self.buffer.clear();
        self.last_rms = 0.0;
    }

    /// Returns estimated latency in milliseconds based on buffer size.
    pub fn latency_ms(&self) -> f32 {
        (self.analysis_size as f32 / self.sample_rate as f32) * 1000.0
    }
}

/// Calculate RMS (Root Mean Square) level of audio samples.
fn calculate_rms(samples: &[f32]) -> f32 {
    if samples.is_empty() {
        return 0.0;
    }
    let sum_squares: f32 = samples.iter().map(|&s| s * s).sum();
    (sum_squares / samples.len() as f32).sqrt()
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_pitch_detector_creation() {
        let config = MicConfig::default();
        let detector = PitchDetector::new(44100, &config);
        assert_eq!(detector.sample_rate, 44100);
        assert_eq!(detector.analysis_size, 2048);
    }

    #[test]
    fn test_pitch_detector_latency() {
        let config = MicConfig::default(); // buffer_size = 2048
        let detector = PitchDetector::new(44100, &config);
        let latency = detector.latency_ms();
        // 2048 / 44100 * 1000 = ~46.4ms
        assert!((latency - 46.4).abs() < 1.0);
    }

    #[test]
    fn test_pitch_detector_empty_buffer() {
        let config = MicConfig::default();
        let mut detector = PitchDetector::new(44100, &config);
        // No samples added, should return None
        assert!(detector.detect().is_none());
    }

    #[test]
    fn test_pitch_detector_sine_wave() {
        let config = MicConfig {
            volume_threshold: 0.001,
            confidence_threshold: 0.5,
            buffer_size: 2048,
            ..MicConfig::default()
        };
        let mut detector = PitchDetector::new(44100, &config);

        // Generate a 440Hz sine wave
        let freq = 440.0f32;
        let samples: Vec<f32> = (0..4096)
            .map(|i| (2.0 * std::f32::consts::PI * freq * i as f32 / 44100.0).sin() * 0.5)
            .collect();

        detector.add_samples(&samples);
        let result = detector.detect();

        // Should detect pitch close to 440Hz
        if let Some(pitch) = result {
            assert!((pitch.frequency - 440.0).abs() < 10.0, "Expected ~440Hz, got {}", pitch.frequency);
            assert_eq!(pitch.midi_note, 69); // A4
        }
        // Note: YIN may not detect perfectly on synthetic signals, so result can be None
    }
}

//! Audio capture and pitch detection module.
//!
//! Provides microphone input with real-time pitch detection for
//! audio-to-MIDI conversion.
//!
//! # Architecture
//!
//! ```text
//! [Microphone] -> [AudioCapture] -> [Ring Buffer] -> [PitchDetector] -> [NoteTracker] -> [MIDI Events]
//!                    (audio thread)                     (main thread)
//! ```
//!
//! The audio callback runs on a high-priority thread and must never block.
//! Samples are transferred via lock-free ring buffer to the main thread
//! where pitch detection and note tracking occur.

pub mod config;
pub mod profiles;
pub mod pitch;
pub mod detector;

#[cfg(not(target_arch = "wasm32"))]
pub mod capture;

pub use config::MicConfig;
pub use profiles::{MicProfile, builtin_profiles};
pub use pitch::{DetectedPitch, NoteEvent, NoteTracker, freq_to_midi, is_in_voice_range};
pub use detector::PitchDetector;

#[cfg(not(target_arch = "wasm32"))]
pub use capture::{
    AudioCapture, list_audio_inputs, get_input_device, default_input_device,
    calculate_rms, rms_to_db, level_for_display,
};
