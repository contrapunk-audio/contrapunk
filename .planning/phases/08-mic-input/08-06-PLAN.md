```rust
// Mic Settings section (show when mic is selected or always in settings)
if self.state.input_port == Some(INPUT_MICROPHONE) {
    ui.add_space(10.0);
    ui.separator();
    ui.add_space(5.0);

    ui.horizontal(|ui| {
        ui.label(egui::RichText::new("MIC SETTINGS").color(GOLD).strong());
    });

    // Profile selector
    ui.horizontal(|ui| {
        ui.label("Profile:");
        let profile_name = self.mic_state.profiles.get(self.mic_state.profile_index)
            .map(|p| p.name.as_str())
            .unwrap_or("Unknown");
        egui::ComboBox::from_id_salt("mic_profile")
            .selected_text(profile_name)
            .show_ui(ui, |ui| {
                for (i, profile) in self.mic_state.profiles.iter().enumerate() {
                    if ui.selectable_label(i == self.mic_state.profile_index, &profile.name).clicked() {
                        self.mic_state.profile_index = i;
                        self.mic_state.config = profile.config.clone();
                        // Update note tracker with new hysteresis settings
                        self.mic_state.note_tracker = NoteTracker::new(
                            self.mic_state.config.hysteresis_cents,
                            self.mic_state.config.min_note_duration_ms,
                        );
                    }
                }
            });
    });

    // Volume threshold slider
    ui.horizontal(|ui| {
        ui.label("Volume Gate:");
        let mut vol = self.mic_state.config.volume_threshold;
        if ui.add(egui::Slider::new(&mut vol, 0.001..=0.2)
            .logarithmic(true)
            .text("threshold")).changed() {
            self.mic_state.config.volume_threshold = vol;
        }
    });

    // Confidence threshold slider
    ui.horizontal(|ui| {
        ui.label("Confidence:");
        let mut conf = self.mic_state.config.confidence_threshold;
        if ui.add(egui::Slider::new(&mut conf, 0.3..=0.95)
            .text("threshold")).changed() {
            self.mic_state.config.confidence_threshold = conf;
        }
    });

    // Latency / buffer size selector
    ui.horizontal(|ui| {
        ui.label("Latency:");
        let sizes = [1024usize, 2048, 4096];
        let labels = ["~23ms (fast)", "~46ms (balanced)", "~93ms (accurate)"];
        let current_idx = sizes.iter().position(|&s| s == self.mic_state.config.buffer_size).unwrap_or(1);

        egui::ComboBox::from_id_salt("mic_latency")
            .selected_text(labels[current_idx])
            .show_ui(ui, |ui| {
                for (i, &label) in labels.iter().enumerate() {
                    if ui.selectable_label(i == current_idx, label).clicked() {
                        self.mic_state.config.buffer_size = sizes[i];
                    }
                }
            });
    });

    // Hysteresis for vibrato smoothing
    ui.horizontal(|ui| {
        ui.label("Vibrato Smoothing:");
        let mut hyst = self.mic_state.config.hysteresis_cents as f32;
        if ui.add(egui::Slider::new(&mut hyst, 20.0..=60.0)
            .text("cents")).changed() {
            self.mic_state.config.hysteresis_cents = hyst as i8;
        }
    });

    // Show current latency in ms
    let latency_ms = self.mic_state.config.buffer_size as f32 / 44100.0 * 1000.0;
    ui.label(egui::RichText::new(format!("Estimated latency: {:.1}ms", latency_ms))
        .color(TEXT_DIM).small());
}
```

```rust
// Show mic status panel when mic is selected as input
if self.state.input_port == Some(INPUT_MICROPHONE) {
    ui.add_space(4.0);
    card(ui, |ui| {
        card_header(ui, "Mic Input");

        ui.horizontal(|ui| {
            // Level meter (vertical bar)
            let rms = self.mic_state.current_rms;
            // Convert RMS to display level (0.0-1.0)
            let level = if rms > 0.0 {
                ((rms.log10() * 20.0 + 60.0) / 60.0).clamp(0.0, 1.0)
            } else {
                0.0
            };
            let meter_height = 50.0;
            let meter_width = 10.0;

            let (rect, _) = ui.allocate_exact_size(
                egui::vec2(meter_width, meter_height),
                egui::Sense::hover(),
            );

            // Background
            ui.painter().rect_filled(rect, 2.0, egui::Color32::from_rgb(20, 20, 25));

            // Level fill (green/yellow/red gradient based on level)
            let fill_height = meter_height * level;
            let fill_rect = egui::Rect::from_min_size(
                egui::pos2(rect.min.x, rect.max.y - fill_height),
                egui::vec2(meter_width, fill_height),
            );
            let fill_color = if level > 0.9 {
                egui::Color32::from_rgb(255, 80, 80) // Red (clipping)
            } else if level > 0.7 {
                egui::Color32::from_rgb(255, 200, 80) // Yellow (hot)
            } else {
                egui::Color32::from_rgb(80, 200, 80) // Green (good)
            };
            ui.painter().rect_filled(fill_rect, 2.0, fill_color);

            ui.add_space(10.0);

            // Pitch display (tuner style)
            ui.vertical(|ui| {
                if let Some(ref pitch) = self.mic_state.last_pitch {
                    // Note name + octave (large) - midi_to_name is already imported
                    let note_name = midi_to_name(pitch.midi_note);
                    ui.label(egui::RichText::new(&note_name)
                        .size(24.0)
                        .strong()
                        .color(egui::Color32::from_rgb(100, 220, 255))); // Cyan for mic

                    // Cents deviation
                    let cents_str = if pitch.cents_deviation >= 0 {
                        format!("+{}c", pitch.cents_deviation)
                    } else {
                        format!("{}c", pitch.cents_deviation)
                    };
                    let cents_color = if pitch.cents_deviation.abs() <= 10 {
                        egui::Color32::from_rgb(80, 200, 80) // Green (in tune)
                    } else if pitch.cents_deviation.abs() <= 25 {
                        egui::Color32::from_rgb(255, 200, 80) // Yellow
                    } else {
                        egui::Color32::from_rgb(255, 100, 100) // Red
                    };
                    ui.label(egui::RichText::new(&cents_str)
                        .size(12.0)
                        .color(cents_color));

                    // Frequency in Hz (small)
                    ui.label(egui::RichText::new(format!("{:.1} Hz", pitch.frequency))
                        .size(10.0)
                        .color(TEXT_DIM));
                } else {
                    // No pitch detected
                    ui.label(egui::RichText::new("---")
                        .size(24.0)
                        .color(TEXT_DIM));
                    ui.label(egui::RichText::new("Listening...")
                        .size(10.0)
                        .color(TEXT_DIM));
                }
            });
        });

        // Permission status if any
        if let Some(ref status) = self.mic_state.permission_status {
            ui.label(egui::RichText::new(status).size(10.0).color(TEXT_SECONDARY));
        }
    });
}
```
